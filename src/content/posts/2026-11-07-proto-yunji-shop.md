<script>
  (function () {
    if (window.__shopViewerInit) return;
    window.__shopViewerInit = true;

    var BASE = "/assets/library/notes/proto-yunji-shop/decks";
    var DECKS = {
      q1: { dir: "q1plan", n: 6, title: "Q1 规划" },
      report: { dir: "exp-report", n: 12, title: "增长实验报告" }
    };

    function setup(viewer) {
      var holder = viewer.parentNode.querySelector("[data-shop-data]");
      var sheets = holder ? JSON.parse(holder.textContent).sheets : [];

      var tabs = [].slice.call(viewer.querySelectorAll("[data-shop-tab]"));
      var panelData = viewer.querySelector('[data-shop-panel="data"]');
      var panelDeck = viewer.querySelector('[data-shop-panel="deck"]');
      var sheetBar = viewer.querySelector("[data-shop-sheets]");
      var table = viewer.querySelector("[data-shop-table]");
      var chartBox = viewer.querySelector("[data-shop-chart]");
      var tableBox = viewer.querySelector("[data-shop-scroll]");
      var dataHint = viewer.querySelector("[data-shop-datahint]");
      var img = viewer.querySelector("[data-shop-img]");
      var countEl = viewer.querySelector("[data-shop-count]");
      var titleEl = viewer.querySelector("[data-shop-title]");
      var prev = viewer.querySelector("[data-shop-prev]");
      var next = viewer.querySelector("[data-shop-next]");

      var mode = "data";
      var sheetIdx = 0;
      var page = 0;

      function renderTable() {
        var sheet = sheets[sheetIdx];
        table.textContent = "";
        if (!sheet) return;
        var thead = document.createElement("thead");
        var trh = document.createElement("tr");
        sheet.rows[0].forEach(function (c) {
          var th = document.createElement("th");
          th.textContent = c;
          trh.appendChild(th);
        });
        thead.appendChild(trh);
        table.appendChild(thead);
        var tbody = document.createElement("tbody");
        for (var i = 1; i < sheet.rows.length; i++) {
          var tr = document.createElement("tr");
          sheet.rows[i].forEach(function (c) {
            var td = document.createElement("td");
            td.textContent = c;
            tr.appendChild(td);
          });
          tbody.appendChild(tr);
        }
        table.appendChild(tbody);
        if (dataHint) {
          dataHint.textContent = sheet.name + " · 共 " + (sheet.rows.length - 1) + " 行 × " + sheet.rows[0].length + " 列"
            + (sheetIdx === 0 ? " · 图形为主，明细可切换" : "");
        }
        // 第一张表（小店整体）用图表呈现，其余仍看明细表
        var showChart = sheetIdx === 0;
        if (chartBox) chartBox.hidden = !showChart;
        if (tableBox) tableBox.hidden = showChart;
        [].forEach.call(sheetBar.querySelectorAll(".shop-sheet"), function (b, i) {
          b.classList.toggle("is-active", i === sheetIdx);
        });
      }

      function renderDeck() {
        var d = DECKS[mode];
        if (!d) return;
        img.src = BASE + "/" + d.dir + "/" + String(page + 1).padStart(2, "0") + ".webp";
        countEl.textContent = (page + 1) + " / " + d.n;
        if (titleEl) titleEl.textContent = d.title;
        prev.disabled = page === 0;
        next.disabled = page === d.n - 1;
      }

      function setMode(next) {
        mode = next;
        tabs.forEach(function (t) {
          t.classList.toggle("is-active", t.dataset.shopTab === mode);
        });
        var isData = mode === "data";
        panelData.hidden = !isData;
        panelDeck.hidden = isData;
        if (isData) renderTable();
        else { page = 0; renderDeck(); }
      }

      sheets.forEach(function (s, i) {
        var b = document.createElement("button");
        b.type = "button";
        b.className = "shop-sheet" + (i === 0 ? " is-active" : "");
        b.textContent = s.name;
        b.addEventListener("click", function () { sheetIdx = i; renderTable(); });
        sheetBar.appendChild(b);
      });

      tabs.forEach(function (t) {
        t.addEventListener("click", function () { setMode(t.dataset.shopTab); });
      });
      prev.addEventListener("click", function () { if (page > 0) { page--; renderDeck(); } });
      next.addEventListener("click", function () { if (page < DECKS[mode].n - 1) { page++; renderDeck(); } });
      img.addEventListener("click", function (e) {
        var r = img.getBoundingClientRect();
        var right = e.clientX - r.left > r.width / 2;
        if (right && page < DECKS[mode].n - 1) { page++; renderDeck(); }
        else if (!right && page > 0) { page--; renderDeck(); }
      });
      viewer.addEventListener("keydown", function (e) {
        if (mode === "data") return;
        if (e.key === "ArrowLeft" && page > 0) { page--; renderDeck(); e.preventDefault(); }
        if (e.key === "ArrowRight" && page < DECKS[mode].n - 1) { page++; renderDeck(); e.preventDefault(); }
      });
      viewer.addEventListener("mousedown", function (e) {
        if (e.target.closest("[data-shop-panel]")) viewer.focus();
      });

      setMode("data");
    }

    function boot() {
      var all = document.querySelectorAll("[data-shop-viewer]");
      for (var i = 0; i < all.length; i++) {
        if (all[i].dataset.shopReady) continue;
        all[i].dataset.shopReady = "1";
        setup(all[i]);
      }
    }

    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", boot);
    } else {
      boot();
    }
    document.addEventListener("swup:contentReplaced", boot);
    document.addEventListener("astro:page-load", boot);
  })();
</script>
